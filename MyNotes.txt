      items.title,

      items.status,

      users.display_name AS owner_name

    JOIN users ON users.id = items.user_id

function searchItems({ q, category, kind }) {
  let sql = `
    SELECT
      items.id,
      items.user_id,
      items.kind,
      items.category,
      items.title,
      items.description,
      items.location,
      items.contact_pref,
      items.status,
      items.created_at,
      users.display_name AS owner_name
    FROM items
    JOIN users ON users.id = items.user_id
    WHERE items.status != 'removed'
  `;

 const user_inputs = []

  if (q) {
    sql += ` AND items.title || ' ' || items.description || ' ' || items.location LIKE ?`;
    user_inputs.push(`%${q}%`);
  }

  if (category && category !== "all") {
    sql += ` AND items.category = ?`;
    user_inputs.push(category);
  }

  if (kind && kind !== "all") {
    sql += ` AND items.kind = ?`;
    user_inputs.push(kind);
  }

  sql += " ORDER BY items.created_at DESC LIMIT 50";
  return db.prepare(sql).all();
}
